APA7 Formatting - Code and Back-Story

Creating APA7 tables is a pain, no matter which software or text processor one uses. I am not sure when, but at some point during my Research Master’s I decided to go full-on R-mode and only use R to create publishable tables. Here is a dump of some of the code I developed and reused over the past 2 years.

Baseline APA7 Format

The function below takes a table as input and uses the table formatting packages gt and gtExtras to return an APA7 consistent table as output. Does not automatically italicize certain table content (e.g., the mean abbreviation M in the table header). These additional content formatting operations need to be conducted before or after applying the APA7 formatting function.

An APA table only consists of horizontal lines (a line at the top of the table, a line separating the header from the table content, and a line at the bottom of the table). All column headers are centered and bold. All table content except for the content in the left-most column is left aligned. Note, however, that APA7 allows for table content to be left aligned if it facilitates readability - meaning that the function should be adapted in this case.

In addition to this baseline formatting, the function also includes some additional arguments, depending on the author’s needs. For instance, it is possible to save the table immediately by passing in a string to the argument save_table. This argument makes use of the function gtsave and requires the string to specify the name and desired file type of the saved table (e.g., “apa7_table.png”). Additionally, the one can specify row groupings - for example, specific rows may contain information on items belonging to a specific scale. Here, the grouping argument can be leveraged. Simply supply a named list in which the names in the list refer to the variable groups (i.e., their desired name) and the values in the list refer to the row indices that belong to the group.

gt_apa7 <- function(gt_table, 
                       text_size = 20, 
                       grouping = NULL, # Supply Named List for Grouping
                       grouping_weight = "normal", 
                       save_table = NULL) {
  
  library(purrr)
  library(gt)
  library(gtExtras)
  
  x <- gt_table
  
  # Grouping Must be Named List
  if (!is.null(grouping)) {
    
    if (is.null(names(grouping)) || any(names(grouping) == "")) {
      stop("`grouping` must be a named list like list('Speed' = 1, 'Distance' = 2)")
    }
    
    group_labels <- names(grouping)
    
    # Iterate Over Named List and add Groupings to gt_table Object
    x <- reduce(
      seq_along(group_labels),
      function(table, group_idx) {
        table |> 
          tab_row_group(
            label = group_labels[group_idx], 
            rows = grouping[[group_idx]], 
            id = as.character(group_idx)
          )
      },
      .init = x
    ) |> 
      row_group_order(groups = as.character(seq_along(group_labels)))
  }
  
  # Baseline Styling
  x <- x |> 
    tab_options(table.font.names = "Times New Roman") |>
    tab_options(
      table_body.hlines.width = px(0),
      column_labels.border.top.color = "black",    
      column_labels.border.bottom.color = "black",
      table_body.border.bottom.color = "black") |> 
    cols_align(align = "center", columns = everything()) |>
    cols_align(align = "left", columns = 1) |> 
    tab_style(
      style = cell_text(weight = "bold", align = "center"),
      locations = cells_column_labels())
  
  # Extra Grouping Styling
  if (!is.null(grouping)) {
    x <- x |> 
      tab_options(row_group.border.top.width = px(0)) |> 
      tab_style(
        style = list(
          cell_text(weight = grouping_weight),
          cell_borders(sides = "bottom", color = "transparent")),
        locations = cells_row_groups()) |> 
      tab_style(
        style = cell_text(indent = px(20)),
        locations = cells_body(columns = 1))
  }
  
  # Final Formatting
  x <- x |> 
    tab_options(table.font.size = px(text_size)) |> 
    tab_options(table.width = pct(100))
  
  # Save if Required
  if (!is.null(save_table)) {
    if (!is.character(save_table)) {
      stop("`save_table` must be the filepath to save the table in string format")
    }
    gtsave(x, filename = save_table)
  }
  
  return(x)
}

Create Table for Descriptive Statistics of Continuous Variable

When writing papers, I developed a preference for a certain format of descriptive statistics tables for continuous variables. Here, the describe() function from the psych package always provides a number of relevant summary statistics that I wanted to display in an APA-consistent format. Therefore, this function makes use of psych::describe() and already formats the table in a way that one must simply use the gp_apa7() function afterward to create an APA7 consistent descriptive statistics table.

gt_describe <- function(df, 
                        distributions = TRUE, 
                        bandwidth = 1, 
                        fixed_axes = TRUE, 
                        statistics = c("n", "mean", "sd", "median", "min", "max")) {
  
  library(gt)
  library(gtExtras)
  library(psych)
  
  descriptives <- df |> 
    psych::describe() |> 
    rownames_to_column(var = "Variable")
  
  if (distributions) {
    dist <- df |>
      summarize(across(everything(), list)) |>
      pivot_longer(cols = everything(), 
                   names_to = "Variable", 
                   values_to = "dist")
    
    table <- descriptives |> 
      left_join(dist, by = "Variable") |>
      gt() |>
      gt_plt_dist( 
        dist,
        type = "histogram", 
        bw = bandwidth, 
        fig_dim = c(5, 15),
        fill_color = "#40B0A6", 
        line_color = "black", 
        same_limit = fixed_axes
      ) |> 
      cols_hide(columns = -c(Variable, statistics, dist)) |>
      cols_label(
        n = md("*n*"),
        mean = md("*M*"),
        sd = md("*SD*"), 
        median = "Median", 
        min = "Min", 
        max = "Max", 
        skew = "Skew", 
        kurtosis = "Kurtosis", 
        dist = "Distribution"
      ) |> 
      fmt_number(
        columns = c(mean, sd, skew, kurtosis, se, median, min, max),
        decimals = 2)
  } else {
    table <- descriptives |> 
      gt() |> 
      cols_hide(columns = -c(Variable, statistics)) |>
      cols_label(
        n = md("*n*"),
        mean = md("*M*"),
        sd = md("*SD*"), 
        median = "Median", 
        min = "Min", 
        max = "Max", 
        skew = "Skew", 
        kurtosis = "Kurtosis"
      ) |> 
      fmt_number(
        columns = c(mean, sd, skew, kurtosis, se, median, min, max),
        decimals = 2)
  }
  
  return(table)
}

Create Correlation Table

Similarly, I wanted to facilitate creating correlation tables. Just use this function on a table of continuous variables and the output supplies a nicely formatted correlation table that only requires the gt_apa7() function as a final formatting step.

gt_cor <- function(df, 
                   custom_names = NULL) {
  
  library(gt)
  
  # As Default, Take the Names Supplied by Data Frame
  names_vars <- names(df)
  numbered_names <- paste0(seq_along(names_vars), ". ", names_vars)
  numbers_only <- paste0(seq_along(names_vars))
  
  # If Provided, Use Custom Names for Variables
  if (!is.null(custom_names)){
    names_vars <- custom_names
    
    numbered_names <- paste0(seq_along(names_vars), ". ", names_vars)
    numbers_only <- paste0(seq_along(names_vars))
  }
  
  matrix_df <- tibble(names = numbered_names)
  
  matrix_names_new_add <- c("Rows/Cols", numbers_only)
  
  correlations <- df |>
    cor()
  
  correlations <- as.matrix(correlations)
  
  correlations_lower <- correlations
  
  correlations_lower[upper.tri(correlations)] <- NA
  
  bivariate_matrix_df <- as.data.frame(correlations_lower) 
  bivariate_matrix_df <- bind_cols(matrix_df, bivariate_matrix_df)
  
  colnames(bivariate_matrix_df) <- matrix_names_new_add
  
  x <- bivariate_matrix_df |> 
    gt() |> 
    sub_missing(missing_text = "") |> 
    fmt_number(columns = c(numbers_only), decimals = 3)
  
  return(x)
  
}
